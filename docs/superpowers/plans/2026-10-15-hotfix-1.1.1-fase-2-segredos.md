**Tech Stack:** Node `>=22.18` (ESM, `node --test`, `node:assert/strict`, `node:fs|path|os|child_process`), zero dependências. Ferramentas chamadas pelo replay: `grep`, `find`, `git`, `rg`/`ag`/`ack`, `ls`, `fd` — sempre `spawnSync(arquivo, [args], { shell: false, timeout: 3000 })`.

- `anotarDiretorios(analise, ctx, dirsIniciais = [ctx.cwd])` (A7 da fase 1) é recursiva nas análises internas e **sobrescreve** `seg.dirs` quando chamada de novo; a guarda reanota cada análise interna com `[d]` para cada diretório possível `d` do segmento externo — inclusive `null` (diretório desconhecido), que chega intacto à análise interna.
- `candidatos(seg, ctx)` devolve `abs` já normalizado (um por diretório; `null` = variável não resolvida ou diretório desconhecido), inclui alvos de redirecionamento e exclui texto puro e valores de predicado de `find`; `verificavelViraCandidato(cand, ctx, ehSegredo, expandir)` (A7 da fase 1) chama `ehSegredo(abs)` e, com o 4º argumento, expande glob/chaves do texto verificável por `expandir(palavra, dir)` — a guarda passa `(abs) => ehArquivoSegredo(abs, ctx)` e `(p, d) => expandirGlob(p, d, { home: ctx.home, vars: varsConhecidas(ctx, d) })`.

- **F2-18 — replay nunca reexecuta efeito colateral do usuário.** "O mesmo comando com os mesmos argumentos" (l.86, l.92) vale só quando os argumentos não mudam a ferramenta: `git` com `-c <k=v>`, `--config-env` ou `--exec-path` antes do subcomando (`core.fsmonitor`, `core.pager`, `alias.*` rodariam programa) ⇒ `{ ok:false }` "opção global do git altera a ferramenta — sem replay" ⇒ deny (O e S), em `git grep` (REQ-H3-8) e em produtor `git …` (REQ-H3-9 a); `fd|fdfind` com `-x|--exec|-X|--exec-batch` (ou grupo curto com `x`/`X`) ⇒ `{ ok:false }` "`fd --exec` executa programa" ⇒ deny. Mesma régua de `rg --pre`/`git grep -O` (l.85). Prova por executável falso que cria um arquivo-marca: depois da decisão o arquivo não existe.
- **F2-19 — laço `while|until … read … done < x` (l.91) pelo acréscimo coordenado da T2.** O laço é achado por `Segmento.palavrasControle` (`while|until` abre; `for|while|until|select` aninham; `done` fecha) e o `read` é o comando de algum segmento da condição (entre o `while` e o `do`). `< <(cmd)` ⇒ produtor = 1º segmento de `alvo.internos[0]`; `< arq` ⇒ campos de `arq` lidos pela guarda só se `arq` é arquivo **regular** (até 1 MiB); `/dev/null` ⇒ lista vazia (allow); outro não regular (`/dev/zero`, FIFO, diretório) ⇒ deny "a lista não pôde ser lida" — ler travaria o hook. `{ while read …; done; } < x`: sem `<` no `done`, vale o `<` do segmento `}` que fecha o grupo logo depois. Só `<(…)`/`>(…)` (palavra que começa com `<(`) é substituição de processo; `< $(cmd)` é caminho (F2-20).
- **F2-20 — atribuição de ambiente no prefixo do segmento × replay (spec omissa; alinhada a l.85 "outra variável ⇒ deny sem replay").** O replay (busca, `find`, produtor) só aceita atribuições cujo nome está em `CI|NODE_ENV|FORCE_COLOR|NO_COLOR|TZ|LANG|LC_ALL` (as "sempre aceitas" das Global Constraints do mestre) e as repassa ao `env` do replay; qualquer outra (`GIT_DIR=`, `GIT_WORK_TREE=`, `RIPGREP_CONFIG_PATH=`…) ⇒ `{ ok:false, motivo:'atribuição de ambiente no segmento altera a ferramenta — sem replay' }` ⇒ deny (O e S): descartá-la faria o replay listar outra coisa que o comando real (`GIT_DIR=<outro>/.git … git grep --untracked` imprime `<outro>/.env`). Laço `done < "$LISTA"` ou `< $(cmd)` (caminho que `normalizar` não resolve) ⇒ deny "laço `while read` alimentado por caminho com variável desconhecida — sem replay"; `< $HOME/lista.txt` resolve e segue a regra do arquivo. `fd`: num grupo curto, a 1ª letra que leva valor (`d E t e S o c j`, fd(1)) encerra o grupo — `-tx` é tipo "executável", `-Hx` é `--exec`.

function mascaraLivre(cru) {
  const livre = new Array(cru.length).fill(false);
  let estado = 'n';
  for (let i = 0; i < cru.length; i++) {
    const c = cru[i];

      if (c === "'") { estado = 's'; continue; }
      if (c === '"') { estado = 'd'; continue; }
      livre[i] = true;
    } else if (estado === 's') {
      if (c === "'") estado = 'n';
    } else {

  }

}

// Expansão de chaves (antes do glob, como o bash); só chaves livres com vírgula de nível 0.
function expandirChaves(cru) {

  for (let i = 0; i < cru.length; i++) {
    if (cru[i] !== '{' || !livre[i]) continue;
    let prof = 0;
    const virgulas = [];
    for (let j = i + 1; j < cru.length; j++) {
      if (!livre[j]) continue;
      if (cru[j] === '{') prof++;
      else if (cru[j] === '}') {
        if (prof === 0) {
          if (!virgulas.length) break;
          const partes = [];
          let ini = i + 1;
          for (const v of virgulas) { partes.push(cru.slice(ini, v)); ini = v + 1; }

          const pre = cru.slice(0, i);
          const pos = cru.slice(j + 1);
          return partes.flatMap((p) => expandirChaves(pre + p + pos));
        }
        prof--;
      } else if (cru[j] === ',' && prof === 0) virgulas.push(j);
    }
  }
  return [cru];
}

// Converte o texto cru (uma alternativa já sem chaves) em peças: {t:'lit',v} | {t:'*'} | {t:'?'} | {t:'cls',v} | {t:'/'}.
// Devolve null quando há variável fora de `vars` ou `$'…'` (não expandimos às cegas).
function pecas(cru, vars, home) {
  const out = [];
  const lit = (v) => { const u = out[out.length - 1]; if (u && u.t === 'lit') u.v += v; else out.push({ t: 'lit', v }); };
  const variavel = (i) => {
    const m = /^\$(?:\{([A-Za-z_][A-Za-z0-9_]*)\}|([A-Za-z_][A-Za-z0-9_]*))/.exec(cru.slice(i));
    if (!m) return null;
    const nome = m[1] || m[2];
    if (!(nome in vars)) return null;
    return { valor: vars[nome], fim: i + m[0].length };
  };
  let i = 0;
  let estado = 'n';
  if (cru.startsWith("$'")) return null;
  if ((cru === '~' || cru.startsWith('~/')) && home) { lit(home); i = 1; }
  while (i < cru.length) {
    const c = cru[i];
    if (estado === 'n') {
      if (c === '\\') { if (i + 1 < cru.length) lit(cru[i + 1]); i += 2; continue; }
      if (c === "'") { estado = 's'; i++; continue; }
      if (c === '"') { estado = 'd'; i++; continue; }
      if (c === '$') { const v = variavel(i); if (!v) return null; for (const ch of v.valor) ch === '/' ? out.push({ t: '/' }) : lit(ch); i = v.fim; continue; }
      if (c === '*') { out.push({ t: '*' }); i++; continue; }
      if (c === '?') { out.push({ t: '?' }); i++; continue; }
      if (c === '[') {
        const f = cru.indexOf(']', i + 2);
        if (f > 0) { out.push({ t: 'cls', v: cru.slice(i, f + 1) }); i = f + 1; continue; }
        lit('['); i++; continue;
      }
      if (c === '/') { out.push({ t: '/' }); i++; continue; }
      lit(c); i++;
    } else if (estado === 's') {
      if (c === "'") { estado = 'n'; i++; continue; }
      c === '/' ? out.push({ t: '/' }) : lit(c); i++;
    } else {
      if (c === '\\' && /["\\$`]/.test(cru[i + 1] || '')) { lit(cru[i + 1]); i += 2; continue; }
      if (c === '"') { estado = 'n'; i++; continue; }

    }
  }

}

const escRe = (s) => s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

}

}

/**
 * Expansão de palavra como o bash sem dotglob, devolvendo as PALAVRAS resultantes (relativas como foram escritas):
 * chaves antes; `*`/`?`/`[…]` livres não casam `.` inicial; sem casamento ⇒ a palavra literal (sem aspas).
 * `~` inicial livre ⇒ `home`; `$NOME`/`${NOME}` de `vars` expandidos; outra variável ou `$'…'` ⇒ [valor] sem expandir.
 */
export function expandirPalavra(palavra, dir, { home = homedir(), vars = {} } = {}) {
  const cru = typeof palavra?.cru === 'string' ? palavra.cru : String(palavra?.valor ?? '');
  const saida = [];
  for (const alt of expandirChaves(cru)) {
    const ps = pecas(alt, { PWD: dir, ...vars }, home);
    if (!ps) { saida.push(String(palavra?.valor ?? alt)); continue; }
    const textoLiteral = ps.map((p) => (p.t === 'lit' ? p.v : p.t === '/' ? '/' : p.t === 'cls' ? p.v : p.t)).join('');
    if (!ps.some((p) => p.t === '*' || p.t === '?' || p.t === 'cls')) { saida.push(textoLiteral); continue; }
    const comps = [[]];
    for (const p of ps) (p.t === '/' ? comps.push([]) : comps[comps.length - 1].push(p));
    let bases = [ps[0]?.t === '/' ? '/' : ''];
    for (const comp of comps) {
      if (!comp.length) continue;
      if (!comp.some((p) => p.t !== 'lit')) { bases = bases.map((b) => juntar(b, comp[0].v)); continue; }
      const re = componenteRe(comp);
      const pontoLiteral = comp[0].t === 'lit' && comp[0].v.startsWith('.');
      const novas = [];
      for (const b of bases) {
        const fisico = resolve(dir, b || '.');
        if (!ehDir(fisico)) continue;
        let nomes = [];
        try { nomes = readdirSync(fisico); } catch { continue; }
        for (const n of nomes.sort()) {
          if (n.startsWith('.') && !pontoLiteral) continue;
          if (re.test(n)) novas.push(juntar(b, n));
        }
      }
      bases = novas;
    }
    const achados = bases.filter((b) => existe(resolve(dir, b)));
    if (achados.length) saida.push(...achados);
    else saida.push(textoLiteral);
  }
  return saida;
}

    const inst = ps.map((p) => (p.t === 'lit' ? p.v : p.t === '?' ? 'x' : p.t === 'cls' ? (p.v.replace(/^\[!?/, '')[0] || 'x') : '')).join('');
    if (ehNomeSegredo(semCaixa ? inst.toLowerCase() : inst)) return true;
  }
  return false;
}

const RE_REF = /\$\{?([A-Za-z0-9_]+)/g;
function refSegredo(texto, extras = new Set()) {
  for (const m of String(texto).matchAll(RE_REF)) if (RE_NOME_SEGREDO.test(m[1]) || extras.has(m[1])) return true;
  return false;
}
const nomeBase = (v) => basename(String(v || ''));

// Variáveis contaminadas: `X=$<NOME>` (atribuição inicial ou `export X=$<NOME>`) em qualquer segmento.
function contaminadas(analise) {
  const nomes = new Set();
  for (const s of analise?.segmentos || []) {
    for (const a of s.atribuicoes || []) if (refSegredo(a.valor?.cru ?? a.valor?.valor ?? '')) nomes.add(a.nome);
    const cmd = nomeBase(s.comando);
    if (cmd === 'export' || cmd === 'declare' || cmd === 'typeset' || cmd === 'local' || cmd === 'readonly') {
      for (const p of (s.palavras || []).slice((s.idxComando ?? 0) + 1)) {
        const m = /^([A-Za-z_][A-Za-z0-9_]*)=(.*)$/s.exec(p.cru ?? p.valor);
        if (m && refSegredo(m[2])) nomes.add(m[1]);
      }
    }
  }
  return nomes;
}

/** REQ-H3-3: devolve o motivo (só nomes de comando/variável) ou null. `analise` dá o contexto do comando inteiro. */
export function exibidor(seg, analise) {
  for (const h of seg.heredocs || []) {
    if ((h.hereString || !h.citado) && refSegredo(h.corpo || '')) return 'here-string/heredoc expande variável de segredo';
  }
  const pal = seg.palavras || [];
  let i = seg.idxComando;
  if (i == null || i < 0 || !pal[i]) return null;
  let cmd = nomeBase(pal[i].valor);
  while ((cmd === 'command' || cmd === 'builtin') && pal[i + 1]) { i++; cmd = nomeBase(pal[i].valor); }
  const resto = pal.slice(i + 1);
  const args = resto.map((p) => p.valor);
  const naoOpc = args.filter((a) => !a.startsWith('-'));
  const soP = args.every((a) => a === '-p');

  switch (cmd) {
    case 'echo': case 'printf': {
      const sujas = contaminadas(analise);
      for (const p of resto) if ((p.marcas?.expansao || p.marcas?.substituicao) && refSegredo(p.cru ?? p.valor, sujas)) return '`echo`/`printf` de variável de segredo';
      return null;
    }
    case 'printenv':
      if (!naoOpc.length) return '`printenv` exibe o ambiente';
      return naoOpc.some((a) => RE_NOME_SEGREDO.test(a)) ? '`printenv` de variável de segredo' : null;
    case 'env': {
      if (args.every((a) => a === '-0' || a === '-p')) return '`env` exibe o ambiente';
      const primeiro = naoOpc[0];
      return primeiro && !primeiro.includes('=') && RE_NOME_SEGREDO.test(primeiro) ? '`env` de variável de segredo' : null;
    }
    case 'set': return args.length === 0 ? '`set` exibe o ambiente' : null;
    case 'export': case 'declare': case 'typeset':
      if (soP) return `\`${cmd}\` exibe o ambiente`;
      if (args.includes('-p') && naoOpc.some((a) => RE_NOME_SEGREDO.test(a))) return `\`${cmd} -p\` de variável de segredo`;
      return null;
    case 'compgen': return args.includes('-v') ? '`compgen -v` lista variáveis' : null;
    case 'gcloud':
      if (args[0] === 'secrets' && args[1] === 'versions' && args[2] === 'access') return '`gcloud secrets versions access`';
      if (args[0] === 'auth' && args.some((a) => a === 'print-access-token' || a === 'print-identity-token')) return '`gcloud auth print-*-token`';
      return null;
    case 'gh':
      if (args[0] === 'auth' && args[1] === 'token') return '`gh auth token`';
      if (args[0] === 'auth' && args[1] === 'status' && args.some((a) => a === '-t' || a === '--show-token')) return '`gh auth status --show-token`';
      return null;
    case 'security': return /^find-.*-password$/.test(args[0] || '') && args.includes('-w') ? '`security find-*-password -w`' : null;
    case 'awk': case 'gawk': case 'mawk': case 'nawk': return args.some((a) => a.includes('ENVIRON')) ? 'programa de `awk` com `ENVIRON`' : null;
    case 'jq':
      return naoOpc.some((a) => /(^|[^A-Za-z0-9_.$])env(?![A-Za-z0-9_])/.test(a) || a.includes('$ENV')) ? 'programa de `jq` com `env`/`$ENV`' : null;
    case 'ps': {
      for (let k = 0; k < args.length; k++) {
        const a = args[k];
        if (k > 0 && OPCOES_PS_COM_VALOR.has(args[k - 1])) continue;
        if (/^-[A-Za-z]+$/.test(a) && a.includes('E')) return '`ps -E` exibe o ambiente dos processos';
        if (/^[A-Za-z]+$/.test(a) && a.includes('e')) return '`ps e` (BSD) exibe o ambiente dos processos';
      }
      return null;
    }
    case 'launchctl': return args[0] === 'getenv' ? '`launchctl getenv`' : null;
    case 'docker': return args[0] === 'inspect' ? '`docker inspect`' : null;
    case 'supabase':
      if (args[0] === 'status') return '`supabase status`';
      return args[0] === 'projects' && args[1] === 'api-keys' ? '`supabase projects api-keys`' : null;
    case 'vercel': return args[0] === 'env' && args[1] === 'pull' ? '`vercel env pull`' : null;
    case 'claude': return args[0] === 'mcp' && args[1] === 'get' ? '`claude mcp get` (imprime o env do servidor)' : null;
    default: return null;
  }
}

function executavel(p) {
  try { accessSync(p, constants.X_OK); return statSync(p).isFile(); } catch { return false; }

/** M-E: `rg` no PATH; senão o executável do Claude Code com argv0 `rg` (M-E = sim) ou o vendor do pacote (M-E = não). */
export function resolverRg(env, medicoes = MEDICOES) {
  for (const d of String(env?.PATH || '').split(delimiter)) {
    if (d && executavel(join(d, 'rg'))) return { arquivo: join(d, 'rg'), argv0: null };
  }
  const exec = env?.CLAUDE_CODE_EXECPATH;

}

// ── valores das palavras para o replay ───────────────────────────────────────────────────
function expandirConhecidas(valor, vars) {
  let out = '';
  for (let i = 0; i < valor.length; i++) {
    if (valor[i] !== '$') { out += valor[i]; continue; }

  }
  return out;
}

/**

 * (semântica de REQ-H3-1), `~` resolvido. Substituição em qualquer palavra do segmento, ou variável desconhecida ⇒ ok:false.

// F2-20: atribuições no prefixo do segmento que o replay aceita (as "sempre aceitas" das Global Constraints do mestre);
// qualquer outra (`GIT_DIR=`, `GIT_WORK_TREE=`, `RIPGREP_CONFIG_PATH=`…) mudaria o que a ferramenta lê ⇒ sem replay.
const ATRIBUICOES_REPLAY = new Set(['CI', 'NODE_ENV', 'FORCE_COLOR', 'NO_COLOR', 'TZ', 'LANG', 'LC_ALL']);

export function valoresParaReplay(seg, dir, ctx) {
  if ((seg.palavras || []).some((p) => p.marcas?.substituicao)) return { ok: false, motivo: 'substituição de comando no segmento (o hook não reexecuta)' };
  const vars = varsConhecidas(ctx, dir);
  const env = { ...ctx.env };
  for (const a of seg.atribuicoes || []) {
    if (!ATRIBUICOES_REPLAY.has(a.nome)) return { ok: false, motivo: 'atribuição de ambiente no segmento altera a ferramenta — sem replay' };

    const v = m.expansao ? expandirConhecidas(a.valor.valor, vars) : (a.valor?.valor ?? '');

  for (const p of args0(seg)) {

  }

}

// ── o que o replay nunca reexecuta (F2-18): opção que troca a ferramenta ou executa programa ──
const GIT_GLOBAL_COM_VALOR = new Set(['-C', '--git-dir', '--work-tree', '--namespace', '--super-prefix']);

/** `git` com `-c <k=v>`, `--config-env` ou `--exec-path` antes do subcomando ⇒ motivo (core.fsmonitor, alias.*… rodariam programa). */
export function gitGlobalAlteraFerramenta(seg) {

  for (let i = 0; i < a.length; i++) {
    const v = a[i];
    if (/^-c/.test(v) || /^--config-env(=|$)/.test(v) || /^--exec-path(=|$)/.test(v)) {
      return 'opção global do git altera a ferramenta (`-c`, `--config-env`, `--exec-path`) — sem replay';
    }
    if (GIT_GLOBAL_COM_VALOR.has(v)) { i++; continue; }
    if (!v.startsWith('-')) return null;
  }

}

// Opções curtas do fd que levam valor (fd(1): -d --max-depth, -E --exclude, -t --type, -e --extension, -S --size,
// -o --owner, -c --color, -j --threads): num grupo curto, o que vem depois delas é o valor (`-tx` = tipo executável).
const FD_CURTAS_COM_VALOR = new Set('dEteSocj');

/** `fd|fdfind` com `-x|--exec|-X|--exec-batch` (também em grupo curto, `-Hx`): executa programa por arquivo achado. */
export function fdExecuta(seg) {
  const n = nome(seg);
  if (n !== 'fd' && n !== 'fdfind') return false;
  for (const { valor: a } of args0(seg)) {
    if (/^--exec(-batch)?(=|$)/.test(a)) return true;
    if (!/^-[^-]/.test(a)) continue;
    for (let k = 1; k < a.length; k++) {
      if (a[k] === 'x' || a[k] === 'X') return true;
      if (FD_CURTAS_COM_VALOR.has(a[k])) break;          // o resto do grupo é o valor

  return false;
}

function rodar(arquivo, args, dir, ctx, { argv0 = null, env = ctx.env } = {}) {
  const r = spawnSync(arquivo, args, {
    cwd: dir, env, timeout: TIMEOUT_MS, maxBuffer: MAX_BUFFER, encoding: 'utf8', shell: false,
    stdio: ['ignore', 'pipe', 'pipe'], ...(argv0 ? { argv0 } : {}),
  });
  if (r.error?.code === 'ETIMEDOUT' || r.signal) return { ok: false, motivo: 'a listagem passou de 3 s (use `-prune`, `--exclude-dir` ou um diretório menor)' };
  if (r.error) return { ok: false, motivo: `a listagem não rodou (${r.error.code || 'erro'})` };
  return { ok: true, status: r.status, saida: r.stdout || '' };
}

const linhas = (saida) => saida.split(/[\n\0]/).map((l) => l.replace(/\r$/, '')).filter((l) => l !== '');

/** Diretório onde a ferramenta resolve caminhos: `git -C <d>` (cumulativo) e `--work-tree`. */
export function dirEfetivo(seg, dir) {

  const a = args0(seg).map((p) => p.valor);
  let d = dir;
  for (let i = 0; i < a.length; i++) {
    if (a[i] === '-C' && a[i + 1] != null) { d = resolve(d, a[i + 1]); i++; continue; }

}

// ── busca (grep -r, rg, ag, ack, git grep) ───────────────────────────────────────────────
const CURTAS_COM_VALOR = {
  grep: new Set('ABCDdefm'),
  rg: new Set('ABCEefgjMmrTt'),
  ag: new Set('ABCGgm'),
  ack: new Set('ABCgm'),
  'git grep': new Set('ABCefmO'),

const LONGAS_COM_VALOR = new Set(['--regexp', '--file', '--max-count', '--context', '--after-context', '--before-context', '--include',
  '--exclude', '--exclude-dir', '--label', '--directories', '--devices', '--binary-files', '--glob', '--iglob', '--type', '--type-not',

  '--max-filesize', '--path-to-ignore', '--ignore-dir', '--pager', '--open-files-in-pager', '--file-from']);
const CONTEXTO = new Set(['A', 'B', 'C']);

// Percorre as opções devolvendo { curtas:Set, longas:Set, operandos:string[] } (valores de opção não viram operandos).
function lerOpcoes(ferr, valores) {
  const curtas = new Set();
  const longas = new Set();
  const operandos = [];
  const comValor = CURTAS_COM_VALOR[ferr];
  let fim = false;
  for (let i = 0; i < valores.length; i++) {
    const a = valores[i];
    if (fim || a === '-' || !a.startsWith('-')) { operandos.push(a); continue; }
    if (a === '--') { fim = true; continue; }
    if (a.startsWith('--')) {
      const [n] = a.split('=');
      longas.add(n);
      if (!a.includes('=') && LONGAS_COM_VALOR.has(n)) i++;
      continue;
    }
    for (let k = 1; k < a.length; k++) {
      curtas.add(a[k]);
      if (comValor.has(a[k])) { if (k === a.length - 1) i++; break; }
    }
  }
  return { curtas, longas, operandos };
}

function ferramentaDeBusca(seg) {
  const n = nome(seg);
  if (n === 'rg' || n === 'ag' || n === 'ack') return { ferr: n, inicio: 0 };
  if (n === 'grep' || n === 'egrep') return { ferr: 'grep', inicio: 0 };
  if (n === 'git') {
    const a = args0(seg).map((p) => p.valor);
    for (let i = 0; i < a.length; i++) {
      if (a[i] === '-C' || a[i] === '-c' || a[i] === '--git-dir' || a[i] === '--work-tree' || a[i] === '--namespace') { i++; continue; }
      if (a[i].startsWith('-')) continue;
      return a[i] === 'grep' ? { ferr: 'git grep', inicio: i + 1 } : null;
    }
  }
  return null;
}

/**
 * Classifica o segmento como busca de REQ-H3-8. null = não é busca (inclui `grep` sem recursão).
 * soNomes: saída só de nomes/números (allow sem replay, salvo pipe para abridor); semReplay: motivo de deny sem replay;
 * amplia: `rg` com `--hidden|-.|-u*|--no-ignore*` (decide a regra "sem binário").
 */
export function classificarBusca(seg) {
  const fb = ferramentaDeBusca(seg);
  if (!fb) return null;
  const valores = args0(seg).slice(fb.inicio).map((p) => p.valor);
  const { curtas, longas } = lerOpcoes(fb.ferr, valores);
  if (fb.ferr === 'grep') {
    const d = valores.findIndex((v, i) => (v === '-d' && valores[i + 1] === 'recurse') || v === '--directories=recurse');
    if (!(curtas.has('r') || curtas.has('R') || longas.has('--recursive') || d >= 0)) return null;
  }
  let semReplay = null;
  if (fb.ferr === 'rg' && (['--pre', '--pre-glob', '--search-zip'].some((o) => longas.has(o)) || curtas.has('z'))) semReplay = '`rg --pre|--search-zip` executa programa';
  if (fb.ferr === 'git grep' && (curtas.has('O') || longas.has('--open-files-in-pager'))) semReplay = '`git grep -O` executa programa';
  if ((fb.ferr === 'ag' || fb.ferr === 'ack') && longas.has('--pager')) semReplay = `\`${fb.ferr} --pager\` executa programa`;
  if (fb.ferr !== 'grep' && fb.ferr !== 'git grep' && (longas.has('--passthru') || longas.has('--passthrough'))) semReplay = '`--passthru` exibe linhas sem casamento';
  const L = fb.ferr !== 'rg' && curtas.has('L');
  const nomes = curtas.has('l') || L || curtas.has('c') || curtas.has('q') ||
    ['--files-with-matches', '--files-without-match', '--count', '--count-matches', '--quiet', '--files', '--type-list'].some((o) => longas.has(o));
  const forma = curtas.has('o') || longas.has('--only-matching') || [...CONTEXTO].some((c) => curtas.has(c)) ||
    ['--context', '--after-context', '--before-context'].some((o) => longas.has(o));
  const amplia = fb.ferr === 'rg' && (longas.has('--hidden') || curtas.has('.') || curtas.has('u') || [...longas].some((o) => o.startsWith('--no-ignore')));
  return { ferramenta: fb.ferr, inicio: fb.inicio, soNomes: nomes && !forma, semReplay, amplia, contagem: curtas.has('c') || longas.has('--count') || longas.has('--count-matches') };
}

// Remove as opções que só mudam a forma da saída (REQ-H3-8), preservando o resto na ordem.
function semOpcoesDeForma(ferr, valores) {
  const tiraCurtas = new Set(ferr === 'rg' ? ['c', 'o', 'q'] : ['c', 'o', 'q', 'L']);
  const tiraLongas = new Set(['--count', '--count-matches', '--only-matching', '--json', '--quiet', '--stats', '--files-without-match']);
  const comValor = CURTAS_COM_VALOR[ferr];
  const out = [];
  let fim = false;
  for (let i = 0; i < valores.length; i++) {
    const a = valores[i];
    if (fim || a === '-' || !a.startsWith('-')) { out.push(a); continue; }
    if (a === '--') { fim = true; out.push(a); continue; }
    if (a.startsWith('--')) {
      const [n] = a.split('=');
      if (ferr === 'rg' && n === '--replace') { if (!a.includes('=')) i++; continue; }
      if (tiraLongas.has(n)) continue;
      out.push(a);
      if (!a.includes('=') && LONGAS_COM_VALOR.has(n) && i + 1 < valores.length) out.push(valores[++i]);
      continue;
    }
    let novo = '-';
    for (let k = 1; k < a.length; k++) {
      const c = a[k];
      if (ferr === 'rg' && c === 'r') { if (k === a.length - 1) i++; break; } // rg -r <x> = --replace: sai junto com o valor
      if (comValor.has(c)) {
        novo += a.slice(k);
        if (k === a.length - 1 && i + 1 < valores.length) { out.push(novo); out.push(valores[++i]); novo = ''; }
        break;
      }
      if (!tiraCurtas.has(c)) novo += c;
    }
    if (novo && novo !== '-') out.push(novo);
  }
  return out;
}

function lerListagem(saida, dirFerr, { contagem = false } = {}) {
  const caminhos = [];
  for (let l of linhas(saida)) {
    if (contagem) l = l.replace(/:\d+$/, '');
    const abs = resolve(dirFerr, l);
    if (!existe(abs)) return { ok: false, motivo: 'saída da listagem não interpretável (linha que não é caminho existente)' };
    caminhos.push(abs);
  }
  return { ok: true, caminhos };
}

function binarioDe(ferr, ctx) {
  if (ferr === 'rg') return resolverRg(ctx.env) || null;
  if (ferr === 'git grep') return { arquivo: 'git', argv0: null };
  return { arquivo: ferr, argv0: null };
}

function montarBusca(seg, dir, ctx, { comL }) {
  const c = classificarBusca(seg);
  if (!c) return { ok: false, motivo: 'não é busca' };
  if (c.semReplay) return { ok: false, motivo: c.semReplay };
  const g = gitGlobalAlteraFerramenta(seg);
  if (g) return { ok: false, motivo: g };
  const v = valoresParaReplay(seg, dir, ctx);
  if (!v.ok) return v;
  const bin = binarioDe(c.ferramenta, ctx);
  if (!bin) return { ok: false, motivo: 'ripgrep não encontrado para o replay' };
  const prefixo = v.args.slice(0, c.inicio);          // git: opções globais + `grep`
  let resto = v.args.slice(c.inicio);
  const cor = c.ferramenta === 'ag' || c.ferramenta === 'ack' ? ['--nocolor'] : ['--color=never'];
  const extra = c.ferramenta === 'rg' ? ['--no-messages'] : c.ferramenta === 'git grep' ? ['--no-color'] : [];
  const listaArquivos = resto.includes('--files') || resto.includes('--type-list');
  if (comL) resto = semOpcoesDeForma(c.ferramenta, resto);
  const l = comL && !listaArquivos ? ['-l'] : [];
  const args = [...prefixo, ...l, ...cor, ...extra, ...resto];
  return { ok: true, c, bin, args, env: v.env, contagem: !comL && c.contagem };
}

/** REQ-H3-8 — busca: mesmo comando, `-l` logo após o nome (antes de qualquer operando e de `--`), sem opções de forma. */
export function replayBusca(seg, dir, ctx) {
  const m = montarBusca(seg, dir, ctx, { comL: true });
  if (!m.ok) return m;
  const r = rodar(m.bin.arquivo, m.args, dir, ctx, { argv0: m.bin.argv0, env: m.env });
  if (!r.ok) return r;
  if (r.status !== 0 && r.status !== 1) return { ok: false, motivo: `a listagem terminou com erro (saída ${r.status})` };
  return lerListagem(r.saida, dirEfetivo(seg, dir));

/** Busca só-nomes como PRODUTOR de pipe com abridor (REQ-H3-9 a): roda como está (só sem cor) e lê os nomes.

export function replayBuscaComoEsta(seg, dir, ctx) {
  const m = montarBusca(seg, dir, ctx, { comL: false });
  if (!m.ok) return m;

/** Troca cada ação por `-print` na mesma posição; `-print`, `-quit`, `-exit` ficam; sem ação nenhuma ⇒ `-print` no fim. */
export function argsReplayFind(valores) {
  const out = [];
  let havia = false;
  for (let i = 0; i < valores.length; i++) {
    const a = valores[i];
    if (ACOES_FIND.comFim.has(a)) {
      let j = i + 1;
      while (j < valores.length && valores[j] !== ';' && valores[j] !== '+') j++;
      out.push('-print'); havia = true; i = j; continue;
    }
    if (ACOES_FIND.zero.has(a)) { out.push('-print'); havia = true; continue; }
    if (ACOES_FIND.um.has(a)) { out.push('-print'); havia = true; i += 1; continue; }
    if (ACOES_FIND.dois.has(a)) { out.push('-print'); havia = true; i += 2; continue; }
    if (a === '-print') havia = true;
    out.push(a);
  }
  if (!havia) out.push('-print');
  return out;
}

/** REQ-H3-8 — find: mesmos argumentos, ações trocadas por `-print` no lugar. */
export function replayFind(seg, dir, ctx) {
  const v = valoresParaReplay(seg, dir, ctx);
  if (!v.ok) return v;

}

// ── diff -r e diretórios listados por find -exec ─────────────────────────────────────────
function fnmatchRe(padrao) {
  return new RegExp('^' + padrao.split('').map((c) => (c === '*' ? '.*' : c === '?' ? '.' : c.replace(/[.+^${}()|[\]\\]/g, '\\$&'))).join('') + '$');
}

/** Arquivos de segredo sob `dir` (recursivo, sem seguir symlink), descontados os nomes-base que casam `excluir`. */
export function segredosSob(dir, excluir, ctx, { prazo = Date.now() + TIMEOUT_MS } = {}) {
  const res = (excluir || []).map(fnmatchRe);
  const achados = [];
  const pilha = [dir];
  while (pilha.length) {
    if (Date.now() > prazo) throw new ErroReplay('a listagem passou de 3 s (restrinja o diretório)');
    const d = pilha.pop();
    let ents = [];
    try { ents = readdirSync(d, { withFileTypes: true }); } catch { continue; }
    for (const e of ents) {
      if (res.some((r) => r.test(e.name))) continue;
      const p = join(d, e.name);
      if (e.isDirectory()) pilha.push(p);
      else if (ehArquivoSegredo(p, ctx)) achados.push(p);
    }
  }
  return achados;
}

/** `diff -r|--recursive`: null se não é; { breve, excluir, operandos } se é. */
export function classificarDiff(seg) {
  if (nome(seg) !== 'diff') return null;
  const a = args0(seg).map((p) => p.valor);
  let rec = false; let breve = false;
  const excluir = []; const operandos = [];
  let fim = false;
  for (let i = 0; i < a.length; i++) {
    const v = a[i];
    if (fim || !v.startsWith('-') || v === '-') { operandos.push(v); continue; }
    if (v === '--') { fim = true; continue; }
    if (v === '--recursive') { rec = true; continue; }
    if (v === '--brief') { breve = true; continue; }
    if (v.startsWith('--exclude=')) { excluir.push(v.slice(10)); continue; }
    if (v === '--exclude' || v === '-x') { if (a[i + 1] != null) excluir.push(a[++i]); continue; }
    if (v.startsWith('--')) continue;
    for (let k = 1; k < v.length; k++) {
      const c = v[k];
      if (c === 'r') rec = true;
      else if (c === 'q') breve = true;
      else if (c === 'x') { const resto = v.slice(k + 1); if (resto) excluir.push(resto); else if (a[i + 1] != null) excluir.push(a[++i]); break; }
    }
  }
  return rec ? { breve, excluir, operandos } : null;
}

    if (prog[i] === '\\') { const d = prog[i + 1]; i += 2; if (!pularDelimitado(d)) return false; while (/[IM]/.test(prog[i] || '')) i++; return true; }
    return true;

const ABRE_LACO = new Set(['for', 'while', 'until', 'select']);

/**
 * Pipelines da análise: { produtor, consumidores, laco, arquivo, segRedir }.
 * `a | b | c` ⇒ produtor `a`, consumidores [b, c]. Laço `while|until … read … done < x` (l.91) ⇒ laco:true, com
 * produtor = 1º segmento de `<(cmd)` (alvo com marca `substituicao`) ou arquivo = palavra `x` (produtor `cat x`).
 * O laço é lido em `Segmento.palavrasControle` (acréscimo coordenado da T2): palavra de controle nunca é `comando`.
 */
export function pipelines(analise) {
  const segs = analise?.segmentos || [];
  const out = [];
  let atual = null;
  for (const s of segs) {
    if (atual) atual.consumidores.push(s);
    else if (s.seguidoDePipe) atual = { produtor: s, consumidores: [], laco: false, arquivo: null, segRedir: null };
    if (atual && !s.seguidoDePipe) { out.push(atual); atual = null; }
  }
  for (let i = 0; i < segs.length; i++) {
    const ws = controles(segs[i]);
    for (let k = 0; k < ws.length; k++) {
      if (ws[k] !== 'while' && ws[k] !== 'until') continue;
      // o `done` deste laço: for|while|until|select abrem, done fecha; o 1º `do` de nível 0 encerra a condição
      let prof = 0;
      let fim = -1;
      let iDo = -1;
      for (let j = i; j < segs.length && fim < 0; j++) {
        const lista = controles(segs[j]);
        for (let m = j === i ? k + 1 : 0; m < lista.length; m++) {
          const w = lista[m];
          if (ABRE_LACO.has(w)) prof++;
          else if (w === 'do' && prof === 0 && iDo < 0) iDo = j;
          else if (w === 'done') { if (prof === 0) { fim = j; break; } prof--; }
        }
      }
      if (fim < 0) continue;
      const condicao = segs.slice(i, iDo < 0 ? fim : iDo);
      if (!condicao.some((s) => nome(s) === 'read')) continue;
      const entrada = (g) => (g?.redirs || []).find((x) => x.op === '<' && (x.fd == null || x.fd === 0) && x.alvo);
      let segR = segs[fim];
      let r = entrada(segR);
      // `{ while read …; done; } < lista`: o `<` está no `}` que fecha o grupo logo depois do `done`
      if (!r && controles(segs[fim + 1])[0] === '}') { segR = segs[fim + 1]; r = entrada(segR); }
      if (!r) continue;
      const base = { consumidores: segs.slice(i, fim + 1), laco: true, segRedir: segR };
      // `< <(cmd)` (substituição de PROCESSO: a palavra começa com `<(`) ⇒ produtor `cmd`; `< arq`, `< "$X"`, `< $(cmd)` ⇒ arquivo
      if (/^[<>]\(/.test(r.alvo.valor ?? '')) out.push({ ...base, produtor: r.alvo.internos?.[0]?.segmentos?.[0] ?? null, arquivo: null });
      else out.push({ ...base, produtor: null, arquivo: r.alvo });
    }
  }
  return out;
}

        if (campos.some((f) => ehArquivoSegredo(resolve(d, f), ctx))) { decisoes.push(negar(ctx, 'laço `while read` abriria arquivo de segredo listado', 'REQ-H3-9')); break; }
      }
      continue;
    }
    const prod = p.produtor;
    const tipo = tipoProdutor(prod);
    if (tipo === 'eco') {
      const pal = (prod.palavras || []).slice((prod.idxComando ?? 0) + 1);
      let desconhecido = false;
      let achou = null;
      for (const d of dirsDe(prod, ctx)) {
        if (d == null) { desconhecido = true; continue; }
        for (const w of pal) {
          if (w.marcas?.substituicao) { desconhecido = true; continue; }
          for (const nomeW of expandirPalavra(w, d, { home: ctx.home, vars: varsConhecidas(ctx, d) })) {
            if (nomeW.includes('$')) { desconhecido = true; continue; }
            if (ehArquivoSegredo(resolve(d, nomeW), ctx)) achou = nomeW;
          }
        }
      }
      if (achou) decisoes.push(negar(ctx, `o pipe abriria ${achou}`, 'REQ-H3-9'));
      else if (desconhecido && ctx.s) decisoes.push(negar(ctx, 'nomes do pipe desconhecidos — liste com `find` antes', 'REQ-H3-9'));
      continue;
    }
    if (tipo === 'repetivel') {
      for (const d of dirsDe(prod, ctx)) {
        if (d == null) { decisoes.push(negar(ctx, 'produtor do pipe em diretório desconhecido', 'REQ-H3-9')); break; }
        const r = replayProdutor(prod, d, ctx);
        if (!r.ok) { decisoes.push(negarMassa(ctx, r.motivo)); break; }
        if (r.campos.some((c) => ehArquivoSegredo(c, ctx))) { decisoes.push(negar(ctx, 'o produtor do pipe lista arquivo de segredo e o consumidor o abriria', 'REQ-H3-9')); break; }
      }
      continue;
    }
    if (ctx.s) decisoes.push(negar(ctx, 'produtor do pipe não é repetível — liste com `find` antes', 'REQ-H3-9'));
    // [O] produtor não repetível: allow (D9, limite §5-4).
  }
  return decisoes;
}
