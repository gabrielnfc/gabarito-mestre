// ───────────────────────────────────────────── runAll (T12, GM-7) — a cadeia REAL do hooks.json, composta (REQ-H0-6)
import { spawnSync as spawnCadeia } from 'node:child_process';
import { readFileSync as lerCadeia } from 'node:fs';
import { join as juntarCadeia, dirname as dirnameCadeia } from 'node:path';
import { fileURLToPath as caminhoCadeia } from 'node:url';
import { compor as comporCadeia } from '../lib/decisao.mjs';

const HOOKS_CADEIA = juntarCadeia(dirnameCadeia(caminhoCadeia(import.meta.url)), '..');
const PLUGIN_CADEIA = juntarCadeia(HOOKS_CADEIA, '..');

function nomeCurtoCadeia(comando) {
  const m = comando.match(/\/hooks\/([a-z-]+\.sh)"(?: ([a-z-]+\.mjs))?$/);
  if (!m) throw new Error(`runAll: comando fora da forma do plugin: ${comando}`);
  return m[2] ?? m[1];
}
function decisaoDoHookCadeia(r) {
  if (r.status === 2) return 'deny';
  if (r.status !== 0) return 'invalido';
  const saida = (r.stdout ?? '').trim();
  if (!saida) return 'allow';
  try {
    const d = JSON.parse(saida)?.hookSpecificOutput?.permissionDecision;
    return d === 'ask' || d === 'deny' ? d : 'allow';
  } catch {
    return 'allow';
  }
}
function motivoDoHookCadeia(h) {
  try {
    const m = JSON.parse(h.stdout)?.hookSpecificOutput?.permissionDecisionReason;
    if (m) return m;
  } catch { /* stdout vazio ou não-JSON: o motivo, se houver, está no stderr */ }
  return h.stderr.trim();
}

export function runAll(matcher, payloadObj, { env = {}, cwd, stdin } = {}) {
  if (payloadObj?.tool_name !== undefined && payloadObj.tool_name !== matcher) {
    throw new Error(`runAll: matcher '${matcher}' ≠ tool_name '${payloadObj.tool_name}'`);
  }
  const grupos = JSON.parse(lerCadeia(juntarCadeia(HOOKS_CADEIA, 'hooks.json'), 'utf8')).hooks.PreToolUse ?? [];
  const entrada = stdin ?? JSON.stringify({ hook_event_name: 'PreToolUse', ...payloadObj, tool_name: matcher });
  // envLimpo (T1, mesmo módulo, fonte única): tira GABARITO_* (hatch, padrões de prod), CLAUDE_PLUGIN_ROOT, CLAUDE_PROJECT_DIR,
  // CLAUDE_CODE_EXECPATH e GIT_DIR/WORK_TREE/INDEX_FILE; preserva GIT_CEILING_DIRECTORIES. `env` do teste vence.
  const ambiente = envLimpo({ CLAUDE_PLUGIN_ROOT: PLUGIN_CADEIA, ...env });
  const dir = cwd ?? payloadObj?.cwd ?? process.cwd();
  const porHook = [];
  for (const g of grupos) {
    if (!new RegExp(`^(?:${g.matcher ? g.matcher : '.*'})$`).test(matcher)) continue;
    for (const h of g.hooks) {
      const comando = h.command.replaceAll('${CLAUDE_PLUGIN_ROOT}', PLUGIN_CADEIA);
      const r = spawnCadeia('bash', ['-c', comando], {
        input: entrada, encoding: 'utf8', cwd: dir, env: ambiente, timeout: (h.timeout ?? 60) * 1000,
      });
      const estourou = r.error?.code === 'ETIMEDOUT';
      porHook.push({
        hook: nomeCurtoCadeia(h.command), code: r.status, stdout: r.stdout ?? '', stderr: r.stderr ?? '',
        decisao: estourou ? 'timeout' : decisaoDoHookCadeia(r), timeout: estourou,
      });
    }
  }
  const final = comporCadeia(porHook.map((h) => ({
    tipo: h.decisao === 'deny' || h.decisao === 'ask' ? h.decisao : 'allow',
    motivo: h.decisao === 'deny' || h.decisao === 'ask' ? motivoDoHookCadeia(h) : '',
    regra: h.hook,
  })));
  return { decisao: final.tipo, motivo: final.motivo, porHook, executados: porHook.map((h) => h.hook) };
}

### Passa a bloquear

**Subagente perde:** escape hatch; `rm` de qualquer tipo; `docker`, `curl`, `gcloud`, `supabase`, `vercel`, `xargs`, `awk`, `tar` e todo comando fora da lista da spec e de `subagente.comandos`; interpretador inline ou por stdin; `source`/`.` (use o binário por caminho, ex.: `.venv/bin/pytest`, ou registre em `subagente.comandos`); `$`/crase fora de aspas simples (`echo "exit $?"`); glob, chaves e `~` fora de aspas (`wc -l docs/*.md` ⇒ `find docs -name "*.md"`); prefixo `NOME=valor` fora de `CI|NODE_ENV|FORCE_COLOR|NO_COLOR|TZ|LANG|LC_ALL` e de `subagente.env`; padrão de busca que é caminho absoluto existente fora do escopo (`grep -rln '/usr/bin/env' bin/` ⇒ `'/usr/bin/en[v]'`); wrappers (`env`, `command`, `builtin`, `exec`, `time`, `nice`, `nohup`, `timeout`, `stdbuf`, `sudo` — REQ-H7-2); `git stash`, `git grep`, `git commit` com opção além de `-m`; `git add` de diretório (só arquivos nomeados); `cd` fora da forma `cd <dir> && …`; subshell `( … )`, grupo `{ …; }` e palavras de controle; escrita fora do próprio worktree e do tmp (o checkout principal é só leitura); `gh` além de leitura de PR/run; MCP de escrita e `execute_sql` em qualquer ambiente, staging inclusive; todo MCP de servidor fora de `subagente.mcp.servidores` (vazio por padrão ⇒ S sem MCP); dev server e infraestrutura sem registro em `subagente.comandos` — porta/schema/namespace chegam por `subagente.env`, e a infraestrutura compartilhada é do orquestrador. Limpeza de contêiner e de scratchpad passa a ser do orquestrador até a 1.2.0.

**Orquestrador perde:** `Read`/`cat` de `.env*` (exceto `segredos.permitidos`) e de credenciais; sobrescrever `.env*` pelo shell (`> .env`; `>>` com literal é permitido, e `Write` também); `docker inspect`, `supabase status`, `supabase projects api-keys`, `vercel env pull` e os demais exibidores; `git add` de diretório que contenha arquivo de segredo não ignorado; escrita via MCP em produção; `execute_sql` que escreve em produção (só leitura pede ask) e `execute_sql` destrutivo sem gatilho de produção — ambos sem hatch, ao contrário do mesmo SQL por `psql` sob R2; edição de settings/hooks/config sem ask. Falsos positivos aceitos: `ls .env* | wc -l` (use `ls .env*` sem pipe); `list_project_custom_environments` (MCP); padrão de busca que é exatamente o caminho de um arquivo de segredo existente (`git grep .env` com `.env` presente ⇒ `git grep -e '\.env'`); busca recursiva ou `find` com ação sobre árvore com segredo não excluído (ex.: `*.pem` de teste versionado ⇒ `--exclude`, `-g '!*.pem'` ou diretório menor); busca ou `find` sem poda que estoura os 3 s do replay ⇒ `-prune`, `--exclude-dir` ou diretório menor; pipeline com abridor cujo produtor repetível ou `echo|printf` lista segredo (`grep -rl API_KEY . | xargs cat`); busca recursiva ou `find` com ação e variável desconhecida ou substituição (`grep -rn "$ID" src/` ⇒ valor literal). Passam: `echo .env.local`, `find … | head`, `grep -rl … | wc -l`, `lsof -ti:3000 | xargs kill`, `git diff | pbcopy`, `curl x | python3 -m json.tool`.

**Ambiente e wrappers (ambos):** variável fora de `CI|NODE_ENV|FORCE_COLOR|NO_COLOR|TZ|LANG|LC_ALL` no prefixo (ou exportada antes no mesmo comando) de busca recursiva (`grep|egrep|fgrep|zgrep|rg|ag|ack|fd|fdfind`), `find`, `git grep`, `git add` amplo e produtor de pipe (REQ-H3-9 (a)) é negada sem replay (`GREP_OPTIONS=-r grep API_KEY .`; `GREP_OPTIONS=-r fgrep API_KEY .`); `GIT_PAGER=cat git log -p` passa; custo declarado: `IFS=: grep -rn x src` nega; `env`, `command`, `builtin`, `exec`, `time`, `nice`, `nohup`, `timeout`, `stdbuf` e `sudo` são desembrulhados pelas guardas de segredo e de configuração (`timeout 5 grep -rn API_KEY .` com `.env` casando é negado; `timeout 5 grep -rn x src` passa); `sudo env`/`timeout 5 env` exibem o ambiente e são negados; opção desconhecida de wrapper é negada.
